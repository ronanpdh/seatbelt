**Architecture:** Starlette app on uvicorn. Clients point their base URL at it. ## What the product is

Seatbelt is an audit harness for AI agents. An organisation runs its employees' agents and assistants through it and gets, for every session, a record that answers who did what, with which model, in what order, and proves it has not been altered since. The record is a hash-chained JSONL ledger signed with Ed25519, verifiable offline with the file, the public key and the CLI. Nothing in it depends on trusting the machine that wrote it or the person who used it.

Today the harness records agents whose code calls the `Recorder` or wraps an SDK client, and it ships an adversarial corpus that attacks an agent and judges the ledger. 0.2.0 adds the piece that makes it an organisational control rather than a developer library: a central gateway that records every client an employee already uses, with no code change, and a report across the fleet.

## Feature summary

Existing (0.1.0):

- **Ledger.** One JSONL file per run. Each event carries the SHA-256 of the previous one and of its own canonical form. Any edit, reorder or deletion is detected and located. Secrets are redacted before hashing.
- **Attestation.** Ed25519-signed manifest per run pins the final hash and file digest, so a rewritten or truncated tail is caught too. Keys never overwritten, private material 0600.
- **Recorder and adapters.** Python API for agents; adapters for the Anthropic SDK (all call styles) and the OpenAI Agents SDK.
- **Policy engine.** Rules at the tool boundary, fail closed, every check recorded.
- **Verification and timeline.** `verify` and `reconstruct`, exit 1 on anything broken, forged or incomplete.
- **Scenario pack.** Adversarial corpus mapped to OWASP Agentic Top 10 and MITRE ATLAS; findings cite event ids. Docker sandbox for targets.
- **Evidence pack.** One signed zip an auditor verifies offline.

New in 0.2.0 (this plan):

- **Recording gateway.** Speaks Anthropic Messages and OpenAI Chat Completions. Serves Claude Code, Claude Desktop and Cowork (via MDM gateway config), Claude SDKs, Codex, Cursor, LangChain, OpenAI SDKs.
- **Identity.** Per-employee issued keys, stored as hashes; principal recorded in every ledger.
- **Sessions.** One ledger per employee session with an idle timeout; optional named runs; crash-safe close on restart.
- **Org policy.** Allowed models, output token cap, denied tools; denials recorded and enforced at the one point a proxy has.
- **Launcher.** `seatbelt run claude` presets the environment so a CLI is recorded without configuration.
- **Fleet report.** Usage by employee, model and tool; denials; failed and unattested runs.
- **Deployment.** Docker image and an MDM profile template for Claude Desktop.

```python
def model_requested(
    self, model: str, request: dict[str, Any], provider: str | None = None
) -> ModelCall:
    """Record the request now; the caller answers with `ModelCall.respond` later."""
    req = self._emit(
        Kind.MODEL_REQUEST,
        self.agent,
        {
            "gen_ai.request.model": model,
            "gen_ai.provider.name": provider,
            "gen_ai.request": request,

@contextmanager
def model_call(
    self, model: str, request: dict[str, Any], provider: str | None = None

    call = self.model_requested(model, request, provider)
    try:
        yield call
    except BaseException as exc:
        if call.response is None:
            call.respond({}, error=_describe(exc))
        raise
```

```python
"""Anthropic Messages wire format on plain JSON: request/response dicts and SSE events."""

from __future__ import annotations

import json
from typing import Any, cast

from seatbelt.ledger.events import Event
from seatbelt.record.recorder import ModelCall, Recorder

KEEP = (
    "messages",
    "system",
    "tools",
    "tool_choice",
    "max_tokens",
    "temperature",
    "thinking",
    "output_config",
    "stop_sequences",
    "betas",
)


class AnthropicFormat:
    provider = "anthropic"

    def __init__(self, rec: Recorder) -> None:
        self._rec = rec
        self._open: dict[str, Event] = {}  # tool_use id -> tool.call event

    @staticmethod
    def model(body: dict[str, Any]) -> str:
        return str(body.get("model", "unknown"))

    def begin(self, body: dict[str, Any]) -> ModelCall:
        """Record tool results carried in the history, then the request."""
        for tool_use_id, content, is_error in tool_results(body):
            call = self._open.pop(tool_use_id, None)
            if call is not None:
                self._rec.tool_returned(call, content, "tool reported error" if is_error else None)
        request = {k: body[k] for k in KEEP if k in body}
        return self._rec.model_requested(self.model(body), request, provider=self.provider)

    def finish(
        self, call: ModelCall, response: dict[str, Any] | None, error: str | None = None
    ) -> list[Event]:
        """Record the response and any tool calls it asks for; returns the tool.call events."""
        if response is None:
            call.respond({}, error=error)
            return []
        usage = cast(dict[str, Any], response.get("usage") or {})  # provider JSON
        answer = call.respond(
            response,
            {k: int(v) for k, v in usage.items() if isinstance(v, int)},
            response_model=cast(str | None, response.get("model")),
            error=error,
        )
        if response.get("stop_reason") is None:
            return []  # abandoned stream: tool inputs may be truncated
        calls: list[Event] = []
        for block in cast(list[dict[str, Any]], response.get("content") or []):
            if block.get("type") == "tool_use":
                event = self._rec.tool_called(
                    str(block.get("name")),
                    cast(dict[str, Any], block.get("input") or {}),
                    call_id=str(block.get("id")),
                    parent_id=answer.id,
                )
                self._open[str(block.get("id"))] = event
                calls.append(event)
        return calls


def tool_results(body: dict[str, Any]) -> list[tuple[str, Any, bool]]:
    """(tool_use_id, content, is_error) for every tool_result block in the request history."""
    out: list[tuple[str, Any, bool]] = []
    for message in cast(list[Any], body.get("messages") or []):
        if not isinstance(message, dict):
            continue
        content = cast(dict[str, Any], message).get("content")
        if not isinstance(content, list):
            continue
        for block in cast(list[Any], content):
            if isinstance(block, dict) and block.get("type") == "tool_result":
                b = cast(dict[str, Any], block)
                out.append((str(b.get("tool_use_id")), b.get("content"), bool(b.get("is_error"))))
    return out


def tool_use_ids(response: dict[str, Any]) -> dict[str, str]:
    """tool_use id -> tool name in a response."""
    return {
        str(b.get("id")): str(b.get("name"))
        for b in cast(list[dict[str, Any]], response.get("content") or [])
        if b.get("type") == "tool_use"
    }


def assemble_sse(events: list[dict[str, Any]]) -> dict[str, Any]:
    """Rebuild the final message from the stream's events, as far as they got."""
    message: dict[str, Any] = {"content": [], "stop_reason": None}
    blocks: list[dict[str, Any]] = []
    partial: dict[int, str] = {}
    for ev in events:
        match ev.get("type"):
            case "message_start":
                message = {**cast(dict[str, Any], ev["message"]), "content": []}
            case "content_block_start":
                blocks.append(dict(cast(dict[str, Any], ev["content_block"])))
            case "content_block_delta":
                i = int(ev["index"])
                delta = cast(dict[str, Any], ev["delta"])
                if delta.get("type") == "text_delta":
                    blocks[i]["text"] = str(blocks[i].get("text", "")) + str(delta["text"])
                elif delta.get("type") == "input_json_delta":
                    partial[i] = partial.get(i, "") + str(delta["partial_json"])
            case "content_block_stop":
                i = int(ev["index"])
                if i in partial:
                    blocks[i]["input"] = json.loads(partial.pop(i) or "{}")
            case "message_delta":
                message.update(cast(dict[str, Any], ev.get("delta") or {}))
                usage = cast(dict[str, Any], message.get("usage") or {})
                message["usage"] = {**usage, **cast(dict[str, Any], ev.get("usage") or {})}
    message["content"] = blocks
    return message
```

```python
class _Recording:
    def __init__(self, rec: Recorder) -> None:
        self._fmt = AnthropicFormat(rec)

    @contextmanager
    def _turn(self, kwargs: dict[str, Any]) -> Generator[list[Message | BetaMessage | None]]:
        """Record tool results and the request; the caller appends the message it got, if any."""
        if kwargs.get("stream"):
            raise TypeError("create(stream=True) is not recorded; use .stream() instead")
        call = self._fmt.begin(kwargs)
        final: list[Message | BetaMessage | None] = []
        try:
            yield final
        except BaseException as exc:
            if call.response is None:
                self._fmt.finish(call, None, error=_describe(exc))
            raise
        response = final[0] if final else None
        if response is None:
            return
        self._fmt.finish(call, response.model_dump(mode="json"))
```

```python
"""OpenAI Chat Completions wire format on plain JSON."""

from __future__ import annotations

import json
from typing import Any, cast

from seatbelt.ledger.events import Event
from seatbelt.record.recorder import ModelCall, Recorder

KEEP = (
    "messages",
    "tools",
    "tool_choice",
    "max_tokens",
    "max_completion_tokens",
    "temperature",
    "reasoning_effort",
    "stop",
)
_USAGE = {"prompt_tokens": "input_tokens", "completion_tokens": "output_tokens"}


def _arguments(
    raw: Any,
) -> dict[str, Any]:  # provider sends a JSON string; keep whatever it is if not
    if isinstance(raw, str):
        try:
            parsed: Any = json.loads(raw or "{}")
        except json.JSONDecodeError:
            return {"_raw": raw}
        return parsed if isinstance(parsed, dict) else {"_raw": raw}
    return cast(dict[str, Any], raw or {})


class OpenAIChatFormat:
    provider = "openai"

    def __init__(self, rec: Recorder) -> None:
        self._rec = rec
        self._open: dict[str, Event] = {}

    @staticmethod
    def model(body: dict[str, Any]) -> str:
        return str(body.get("model", "unknown"))

    def begin(self, body: dict[str, Any]) -> ModelCall:
        for message in cast(list[Any], body.get("messages") or []):
            if isinstance(message, dict) and message.get("role") == "tool":
                m = cast(dict[str, Any], message)
                call = self._open.pop(str(m.get("tool_call_id")), None)
                if call is not None:
                    self._rec.tool_returned(call, m.get("content"))
        request = {k: body[k] for k in KEEP if k in body}
        return self._rec.model_requested(self.model(body), request, provider=self.provider)

    def finish(
        self, call: ModelCall, response: dict[str, Any] | None, error: str | None = None
    ) -> list[Event]:
        if response is None:
            call.respond({}, error=error)
            return []
        raw_usage = cast(dict[str, Any], response.get("usage") or {})
        usage = {
            _USAGE[k]: int(v) for k, v in raw_usage.items() if k in _USAGE and isinstance(v, int)
        }
        answer = call.respond(
            response, usage, response_model=cast(str | None, response.get("model")), error=error
        )
        choices = cast(list[dict[str, Any]], response.get("choices") or [])
        if not choices or choices[0].get("finish_reason") is None:
            return []
        message = cast(dict[str, Any], choices[0].get("message") or {})
        calls: list[Event] = []
        for tc in cast(list[dict[str, Any]], message.get("tool_calls") or []):
            fn = cast(dict[str, Any], tc.get("function") or {})
            event = self._rec.tool_called(
                str(fn.get("name")),
                _arguments(fn.get("arguments")),
                call_id=str(tc.get("id")),
                parent_id=answer.id,
            )
            self._open[str(tc.get("id"))] = event
            calls.append(event)
        return calls


def tool_call_ids(response: dict[str, Any]) -> dict[str, str]:
    choices = cast(list[dict[str, Any]], response.get("choices") or [])
    message = cast(dict[str, Any], choices[0].get("message") or {}) if choices else {}
    return {
        str(tc.get("id")): str(cast(dict[str, Any], tc.get("function") or {}).get("name"))
        for tc in cast(list[dict[str, Any]], message.get("tool_calls") or [])
    }


def tool_results(body: dict[str, Any]) -> list[str]:
    """tool_call ids answered in this request."""
    return [
        str(cast(dict[str, Any], m).get("tool_call_id"))
        for m in cast(list[Any], body.get("messages") or [])
        if isinstance(m, dict) and m.get("role") == "tool"
    ]


def assemble_sse(chunks: list[dict[str, Any]]) -> dict[str, Any]:
    out: dict[str, Any] = {"object": "chat.completion", "choices": []}
    message: dict[str, Any] = {"role": "assistant", "content": None}
    tool_calls: dict[int, dict[str, Any]] = {}
    finish: str | None = None
    for chunk in chunks:
        for key in ("id", "created", "model", "system_fingerprint"):
            if key in chunk:
                out[key] = chunk[key]
        if "usage" in chunk and chunk["usage"]:
            out["usage"] = chunk["usage"]
        for choice in cast(list[dict[str, Any]], chunk.get("choices") or []):
            delta = cast(dict[str, Any], choice.get("delta") or {})
            if delta.get("content"):
                message["content"] = str(message.get("content") or "") + str(delta["content"])
            for tc in cast(list[dict[str, Any]], delta.get("tool_calls") or []):
                i = int(tc.get("index", 0))
                slot = tool_calls.setdefault(
                    i, {"id": None, "type": "function", "function": {"name": "", "arguments": ""}}
                )
                if tc.get("id"):
                    slot["id"] = tc["id"]
                fn = cast(dict[str, Any], tc.get("function") or {})
                if fn.get("name"):
                    slot["function"]["name"] = fn["name"]
                slot["function"]["arguments"] += str(fn.get("arguments") or "")
            if choice.get("finish_reason"):
                finish = str(choice["finish_reason"])
    if tool_calls:
        message["tool_calls"] = [tool_calls[i] for i in sorted(tool_calls)]
    out["choices"] = [{"index": 0, "finish_reason": finish, "message": message}]
    return out
```

```python
"""Gateway configuration: one YAML file, unknown keys rejected, secrets stored as hashes."""

from __future__ import annotations

import hashlib
import secrets
from datetime import date
from pathlib import Path
from typing import Any, cast

import yaml
from pydantic import BaseModel, ConfigDict, Field, ValidationError

_PREFIX = "sbk_"


class _Strict(BaseModel):
    model_config = ConfigDict(extra="forbid")


class Upstream(_Strict):
    url: str
    key_env: str  # name of the env var holding the real provider key; never the key itself


class PolicyConfig(_Strict):
    models: list[str] | None = None
    tools_denied: list[str] = Field(default_factory=list)
    max_output_tokens: int | None = None


class Principal(_Strict):
    id: str
    key_sha256: str
    issued: date


class GatewayConfig(_Strict):
    listen: str = "127.0.0.1:8080"
    signing_key: Path
    ledgers: Path
    session_idle: int = 900  # seconds
    upstreams: dict[str, Upstream]
    policy: PolicyConfig = Field(default_factory=PolicyConfig)
    principals: list[Principal] = Field(default_factory=list)

    def lookup(self, key: str) -> Principal | None:
        digest = key_hash(key)
        return next(
            (p for p in self.principals if secrets.compare_digest(p.key_sha256, digest)), None
        )


def key_hash(key: str) -> str:
    return hashlib.sha256(key.encode()).hexdigest()


def load_config(path: Path) -> GatewayConfig:
    try:
        data: Any = yaml.safe_load(path.read_text(encoding="utf-8"))  # YAML is untyped
        cfg = GatewayConfig.model_validate(data)
    except (yaml.YAMLError, ValidationError, OSError) as exc:
        raise ValueError(f"{path}: {exc}") from exc
    base = path.resolve().parent
    cfg.signing_key = base / cfg.signing_key
    cfg.ledgers = base / cfg.ledgers
    return cfg


def add_principal(path: Path, principal_id: str) -> str:
    """Append a principal, return the one-time key. Rewrites the file, so comments are lost."""
    cfg = load_config(path)
    if any(p.id == principal_id for p in cfg.principals):
        raise ValueError(f"{principal_id} already has a key; delete its line to reissue")
    key = _PREFIX + secrets.token_urlsafe(32)
    raw = cast(
        dict[str, Any], yaml.safe_load(path.read_text(encoding="utf-8"))
    )  # keep the file's own paths
    raw.setdefault("principals", []).append(
        {"id": principal_id, "key_sha256": key_hash(key), "issued": date.today().isoformat()}
    )
    path.write_text(yaml.safe_dump(raw, sort_keys=False), encoding="utf-8")
    return key
```

```python
"""One ledger per employee session. Opened on first request, closed on idle, end, or shutdown."""

from __future__ import annotations

import re
import secrets
import threading
import time
from collections.abc import Callable
from contextlib import ExitStack
from dataclasses import dataclass, field
from datetime import UTC, datetime
from pathlib import Path
from typing import Any

from seatbelt import __version__
from seatbelt.attest.manifest import sidecar
from seatbelt.attest.sign import Signer, attest
from seatbelt.ledger.events import Actor, ActorType, Kind
from seatbelt.ledger.store import Ledger, read_events
from seatbelt.record.recorder import Recorder

_UNSAFE = re.compile(r"[^A-Za-z0-9_-]+")


def _slug(text: str) -> str:
    return _UNSAFE.sub("_", text).strip("_")[:40] or "x"


@dataclass
class Session:
    rec: Recorder
    stack: ExitStack
    lock: threading.Lock = field(default_factory=threading.Lock)
    last: float = 0.0
    formats: dict[str, Any] = field(default_factory=dict[str, Any])  # per-format state, see app.py
    denied_calls: set[str] = field(default_factory=set[str])  # tool call ids policy refused


class Sessions:
    def __init__(
        self,
        root: Path,
        signer: Signer | None,
        idle: float,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        self._root = root
        self._signer = signer
        self._idle = idle
        self._clock = clock
        self._open: dict[tuple[str, str | None], Session] = {}
        self._lock = threading.Lock()
        root.mkdir(parents=True, exist_ok=True)

    def get(self, principal: str, run: str | None, meta: dict[str, Any]) -> Session:
        with self._lock:
            session = self._open.get((principal, run))
            if session is None:
                session = self._start(principal, run, meta)
                self._open[(principal, run)] = session
            session.last = self._clock()
            return session

    def _start(self, principal: str, run: str | None, meta: dict[str, Any]) -> Session:
        stamp = datetime.now(UTC).strftime("%Y%m%dT%H%M%S")
        run_id = f"{_slug(principal)}-{_slug(run) if run else stamp}-{secrets.token_hex(4)}"
        stack = ExitStack()
        rec = stack.enter_context(
            Recorder.start(
                self._root,
                agent_id="gateway",
                agent_version=__version__,
                run_id=run_id,
                metadata={"principal.id": principal, "run.name": run, **meta},
                signer=self._signer,
            )
        )
        return Session(rec=rec, stack=stack, last=self._clock())

    def end(self, principal: str, run: str | None) -> bool:
        with self._lock:
            session = self._open.pop((principal, run), None)
        if session is None:
            return False
        self._close(session)
        return True

    def sweep(self) -> int:
        now = self._clock()
        with self._lock:
            stale = [k for k, s in self._open.items() if now - s.last > self._idle]
            sessions = [self._open.pop(k) for k in stale]
        for s in sessions:
            self._close(s)
        return len(sessions)

    def close_all(self) -> None:
        with self._lock:
            sessions = list(self._open.values())
            self._open.clear()
        for s in sessions:
            self._close(s)

    @staticmethod
    def _close(session: Session) -> None:
        with session.lock:  # never close mid-request
            session.stack.close()


def close_open_chains(root: Path, signer: Signer | None) -> list[Path]:
    """After a crash: append a failed run.end to every open chain and sign it."""
    closed: list[Path] = []
    for path in sorted(root.glob("*.jsonl")):
        events = list(read_events(path))
        if not events or events[-1].kind is Kind.RUN_END:
            continue
        ledger = Ledger(path, events[0].run_id)
        ledger.append(
            Kind.RUN_END,
            Actor(type=ActorType.AGENT, id="gateway", version=__version__),
            {"run.ok": False, "run.error": "gateway restarted", "run.events": len(events) + 1},
        )
        if signer is not None and not sidecar(path).exists():
            attest(path, signer)
        closed.append(path)
    return closed
```

Threading rather than asyncio locks because `Recorder` and `Ledger` are synchronous and the sweeper runs on a thread; the Starlette handlers call into sessions from a worker via `starlette.concurrency.run_in_threadpool` where they block (Task 6).

```python
def _meta(request: Request) -> dict[str, Any]:
    return {
        "client.ip": request.client.host if request.client else None,
        "client.user_agent": request.headers.get("user-agent"),
        "gateway.version": __import__("seatbelt").__version__,
    }


def create_app(
    cfg: GatewayConfig, sessions: Sessions, transport: httpx2.BaseTransport | None = None
) -> Starlette:
    async def relay(request: Request) -> Response:
        principal = _principal(cfg, request)
        if principal is None:
            return JSONResponse({"error": "unknown seatbelt key"}, status_code=401)
        upstream_name, auth_header, format_cls = _FORMATS[request.url.path]
        upstream = cfg.upstreams[upstream_name]
        real_key = os.environ.get(upstream.key_env, "")
        body = cast(dict[str, Any], await request.json())  # provider JSON
        session = sessions.get(principal.id, request.headers.get(RUN_HEADER), _meta(request))
        fmt = session.formats.setdefault(format_cls.provider, format_cls(session.rec))

        def record_request() -> Any:  # ModelCall; kept opaque here
            with session.lock:
                return fmt.begin(body)

        call = await run_in_threadpool(record_request)
        async with httpx2.AsyncClient(
            base_url=upstream.url, transport=app.state.transport, timeout=600
        ) as client:
            resp = await client.post(
                request.url.path,
                json=body,
                headers=_upstream_headers(request, auth_header, real_key),
            )
        ok = resp.status_code < 400
        payload = (
            resp.json()
            if ok and resp.headers.get("content-type", "").startswith("application/json")
            else None
        )
        error = None if ok else f"{resp.status_code}: {resp.text[:200]}"

        def record_response() -> None:
            with session.lock:
                fmt.finish(call, payload, error=error)

        await run_in_threadpool(record_response)
        return Response(
            resp.content, status_code=resp.status_code, media_type=resp.headers.get("content-type")
        )

    async def end_run(request: Request) -> Response:
        principal = _principal(cfg, request)
        if principal is None:
            return JSONResponse({"error": "unknown seatbelt key"}, status_code=401)
        name = request.path_params["name"]
        found = await run_in_threadpool(sessions.end, principal.id, name)
        return Response(status_code=204 if found else 404)

    app = Starlette(
        routes=[
            Route("/v1/messages", relay, methods=["POST"]),
            Route("/v1/chat/completions", relay, methods=["POST"]),
            Route("/seatbelt/runs/{name}/end", end_run, methods=["POST"]),
        ]
    )
    app.state.transport = transport
    return app
```

`session.formats` holds one format instance per provider so open tool calls carry across requests in the session. Per-session `threading.Lock` around every recorder touch keeps the chain ordered even when a client fires requests in parallel.

```python
def _sse_events(raw: bytes) -> list[dict[str, Any]]:
    out: list[dict[str, Any]] = []
    for block in raw.decode(errors="replace").split("\n\n"):
        data = "".join(line[5:].strip() for line in block.splitlines() if line.startswith("data:"))
        if data and data != "[DONE]":

    return out


async def _stream(
    request, session, fmt, call, base_url, headers, body, format_cls
) -> Response:  # annotate fully
    client = httpx2.AsyncClient(
        base_url=base_url, transport=request.app.state.transport, timeout=600
    )
    upstream = await client.send(
        client.build_request("POST", request.url.path, json=body, headers=headers), stream=True
    )
    if upstream.status_code >= 400:
        text = (await upstream.aread()).decode(errors="replace")
        await client.aclose()
        await run_in_threadpool(
            _locked(session, fmt.finish, call, None, f"{upstream.status_code}: {text[:200]}")
        )
        return Response(
            text, status_code=upstream.status_code, media_type=upstream.headers.get("content-type")
        )
    received = bytearray()

    async def body_iter():
        try:
            async for chunk in upstream.aiter_raw():
                received.extend(chunk)
                yield chunk
        finally:
            await upstream.aclose()
            await client.aclose()
            assembled = _ASSEMBLE[format_cls.provider](_sse_events(bytes(received)))
            await run_in_threadpool(_locked(session, fmt.finish, call, assembled))

    return StreamingResponse(
        body_iter(),
        status_code=upstream.status_code,
        media_type=upstream.headers.get("content-type"),
    )


def _locked(session: Session, fn, *args):
    def run():
        with session.lock:
            return fn(*args)

    return run
```

```python
from __future__ import annotations

import threading
import time

import uvicorn

from seatbelt.attest.sign import Signer
from seatbelt.gateway.app import create_app
from seatbelt.gateway.config import GatewayConfig
from seatbelt.gateway.sessions import Sessions, close_open_chains


def serve(cfg: GatewayConfig) -> None:
    signer = Signer.from_file(cfg.signing_key)
    closed = close_open_chains(cfg.ledgers, signer)
    if closed:
        print(f"closed {len(closed)} chains left open by a previous run")
    sessions = Sessions(cfg.ledgers, signer, idle=cfg.session_idle)
    stop = threading.Event()

    def sweeper() -> None:
        while not stop.wait(30):
            sessions.sweep()

    threading.Thread(target=sweeper, daemon=True).start()
    host, _, port = cfg.listen.rpartition(":")
    try:
        uvicorn.run(create_app(cfg, sessions), host=host, port=int(port), log_level="info")
    finally:
        stop.set()
        sessions.close_all()
```

```python
"""`seatbelt run <cli>`: preset the environment so an existing CLI talks to the gateway."""

from __future__ import annotations

import os
import secrets
import subprocess
import tomllib
from pathlib import Path

import httpx2

DEFAULT_CONFIG = Path.home() / ".config" / "seatbelt" / "gateway.toml"

PRESETS: dict[str, tuple[dict[str, str], str]] = {  # cli -> (env template, base url suffix)
    "claude": (
        {
            "ANTHROPIC_BASE_URL": "{url}",
            "ANTHROPIC_AUTH_TOKEN": "{key}",
            "ANTHROPIC_CUSTOM_HEADERS": "X-Seatbelt-Run: {run}",
        },
        "",
    ),
    "codex": ({"OPENAI_BASE_URL": "{url}/v1", "OPENAI_API_KEY": "{key}"}, ""),
}


def load_client_config(path: Path = DEFAULT_CONFIG) -> tuple[str, str]:
    try:
        data = tomllib.loads(path.read_text(encoding="utf-8"))
        return str(data["url"]).rstrip("/"), str(data["key"])
    except (OSError, KeyError, tomllib.TOMLDecodeError) as exc:
        raise ValueError(f"{path}: {exc}; expected url = ... and key = ...") from exc


def environment(cli: str, url: str, key: str, run: str, base: dict[str, str]) -> dict[str, str]:
    if cli not in PRESETS:
        raise ValueError(f"no preset for {cli}; known: {', '.join(PRESETS)}")
    template, _ = PRESETS[cli]
    env = {k: v for k, v in base.items() if k not in ("ANTHROPIC_API_KEY", "OPENAI_API_KEY")}
    env.update({k: v.format(url=url, key=key, run=run) for k, v in template.items()})
    return env


def run_cli(
    cli: str,
    args: list[str],
    config: Path = DEFAULT_CONFIG,
    exe: str | None = None,
    transport: httpx2.BaseTransport | None = None,
) -> int:
    url, key = load_client_config(config)
    run = f"{cli}-{secrets.token_hex(4)}"
    env = environment(cli, url, key, run, dict(os.environ))
    try:
        return subprocess.run([exe or cli, *args], env=env).returncode  # inherits the TTY
    finally:
        with httpx2.Client(base_url=url, transport=transport, timeout=10) as client:
            try:
                client.post(f"/seatbelt/runs/{run}/end", headers={"authorization": f"Bearer {key}"})
            except httpx2.HTTPError:
                pass  # the idle sweeper closes it
```

```python
"""Aggregate a runs directory: who used what, how much, and what was refused."""

from __future__ import annotations

from collections import defaultdict
from pathlib import Path

from pydantic import BaseModel, ConfigDict, Field

from seatbelt.attest.manifest import sidecar
from seatbelt.ledger.events import Kind
from seatbelt.ledger.store import read_events


class Usage(BaseModel):
    model_config = ConfigDict(extra="forbid")
    runs: int = 0
    calls: int = 0
    input_tokens: int = 0
    output_tokens: int = 0
    denials: int = 0


class Fleet(BaseModel):
    model_config = ConfigDict(extra="forbid")
    runs: int = 0
    by_principal: dict[str, Usage] = Field(default_factory=dict)
    by_model: dict[str, Usage] = Field(default_factory=dict)
    by_tool: dict[str, int] = Field(default_factory=dict)
    failed: list[str] = Field(default_factory=list)
    unattested: list[str] = Field(default_factory=list)
    incomplete: list[str] = Field(default_factory=list)


def fleet(runs: Path) -> Fleet:
    out = Fleet()
    tools: dict[str, int] = defaultdict(int)
    for path in sorted(runs.glob("*.jsonl")):
        events = list(read_events(path))
        if not events:
            continue
        out.runs += 1
        principal = str(events[0].attrs.get("principal.id") or "unknown")
        who = out.by_principal.setdefault(principal, Usage())
        who.runs += 1
        for e in events:
            if e.kind is Kind.MODEL_RESPONSE:
                model = str(e.attrs.get("gen_ai.response.model"))
                m = out.by_model.setdefault(model, Usage())
                for u in (who, m):
                    u.calls += 1
                    u.input_tokens += int(e.attrs.get("gen_ai.usage.input_tokens") or 0)
                    u.output_tokens += int(e.attrs.get("gen_ai.usage.output_tokens") or 0)
            elif e.kind is Kind.TOOL_CALL:
                tools[str(e.attrs.get("gen_ai.tool.name"))] += 1
            elif e.kind is Kind.POLICY_CHECK and e.attrs.get("policy.allowed") is False:
                who.denials += 1
        last = events[-1]
        if last.kind is not Kind.RUN_END:
            out.incomplete.append(events[0].run_id)
        elif last.attrs.get("run.ok") is False:
            out.failed.append(events[0].run_id)
        if not sidecar(path).exists():
            out.unattested.append(events[0].run_id)
    out.by_tool = dict(sorted(tools.items()))
    return out
```
